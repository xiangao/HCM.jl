**Goal:** Build the HCM.jl foundation — package skeleton, a hard+soft intervention/estimand framework, and the confounder motif estimated in Turing — validated against analytic ground truth.

**Architecture:** Pure-Julia layers (interventions, estimand, sim, spec) wrap one Turing `@model`. Interventions are transforms of a within-unit treatment distribution; a pure `ate(fit, intervention; baseline)` maps posterior draws → ATE draws (hard and soft, gaussian and bernoulli) without refitting. Mirrors the R `hcm` separation of model from estimand.

**Tech Stack:** Julia, Turing.jl, DataFrames, StatsModels, Distributions, MCMCChains, FixedEffectModels, Documenter, Test.

Exact names/signatures used across tasks:

- `link_inv(η, family::Symbol)` → `family === :gaussian ? η : logistic(η)`. Vectorized via broadcasting.
- Intervention types: `Hard(a_star::Real)`, `Soft(a_star::Real, ε::Real)`.
- `expected_outcome(β0, β1, p, family, intv)` where `β0,β1,p` are `draws × n_units` matrices → vector of length `draws` (per-draw `E[Y; do(intv)]`, averaged over units).
- `ate(β0, β1, p, family, intv; baseline)` → vector of ATE draws = `expected_outcome(intv) .- expected_outcome(baseline)`. Default `baseline`: `Hard(0)` for a `Hard` intv; `Soft(a_star, 0)` for a `Soft` intv.
- `sim_hcm(:confounder; n, m, params=NamedTuple(), family=:gaussian, seed=nothing)` → `NamedTuple` with `data::DataFrame` (cols `:unit,:subunit,:a,:y`), `true_hard::Float64`, `true_soft::Function` (`(a_star,ε)->Float64`), `u::Vector`, `p::Vector`.
- `HCMSpec` struct: fields `outcome::Symbol, treatment::Symbol, unit_id::Vector{Int}, n_units::Int, y::Vector, a::Vector{Int}, family::Symbol, motif::Symbol`.
- `hcm_spec(formula, data; unit, subunit, motif=:confounder, family=:gaussian)` → `HCMSpec`.
- `hcm(formula, data; unit, subunit, motif=:confounder, family=:gaussian, chains=2, iter=600, kwargs...)` → `HCMFit`.
- `HCMFit` struct: `β0::Matrix, β1::Matrix, p::Matrix, spec::HCMSpec, chain` (draws×n_units matrices).
- `ate(fit::HCMFit, intv; baseline=default_baseline(intv))` → vector of ATE draws.
- `summarize_ate(draws)` → `(; mean, std, q025, q975)`.

**Estimand math (spec §3):** Hard `E[Y;do(a★)] = mean_i link_inv(β0_i + β1_i·a★)`. Soft `E[Y;do(σ_{a★,ε})] = mean_i[ (1−ε)(p_i·link_inv(β0_i+β1_i) + (1−p_i)·link_inv(β0_i)) + ε·link_inv(β0_i+β1_i·a★) ]`.

```julia
link_inv(η, family::Symbol) = family === :gaussian ? η :
                              family === :bernoulli ? logistic.(η) :
                              error("unknown family: $family")

# per-draw E[Y; do(intv)], averaged over units. β0,β1,p are draws × n_units.
function expected_outcome(β0, β1, p, family, intv::Hard)
    vec(mean(link_inv(β0 .+ β1 .* intv.a_star, family); dims=2))
end

function expected_outcome(β0, β1, p, family, intv::Soft)
    treated  = link_inv(β0 .+ β1, family)            # A=1
    control  = link_inv(β0, family)                  # A=0
    star     = link_inv(β0 .+ β1 .* intv.a_star, family)
    obs_mix  = p .* treated .+ (1 .- p) .* control    # observed within-unit law
    do_mix   = (1 - intv.ε) .* obs_mix .+ intv.ε .* star
    vec(mean(do_mix; dims=2))
end

function ate(β0, β1, p, family, intv; baseline=default_baseline(intv))
    expected_outcome(β0, β1, p, family, intv) .- expected_outcome(β0, β1, p, family, baseline)
end
```

Note `link_inv` for `:gaussian` returns its argument unchanged (works for scalar and array); for `:bernoulli` it broadcasts `logistic`. In the scalar test `link_inv(0.0, :bernoulli)` returns `logistic.(0.0)` which is a 0-d result `≈ 0.5` — acceptable; if the scalar test errors on the dot-broadcast, change `:bernoulli` branch to `logistic(η)` and rely on callers broadcasting `link_inv.(...)`. Pick whichever makes both the scalar and matrix tests pass; the matrix paths above call `link_inv(matrix, family)` so the `:bernoulli` branch must accept a matrix — keep `logistic.(η)`.

Add to `src/HCM.jl`: `include("estimand.jl")` and `export ate, link_inv`. (`expected_outcome`/`default_baseline` stay internal.)

```julia
function sim_hcm(motif::Symbol; n, m, params=NamedTuple(), family::Symbol=:gaussian, seed=nothing)
    motif === :confounder || error("A1 implements only :confounder")
    seed === nothing || Random.seed!(seed)
    p_ = merge((β0=0.0, β1=0.5, sd_u=1.0, α_p=0.0, ρ=1.0, sd_y=0.3), params)
    u = randn(n) .* p_.sd_u
    prop = logistic.(p_.α_p .+ p_.ρ .* u)                 # propensity p_i
    unit = repeat(1:n, inner=m)
    subunit = repeat(1:m, outer=n)
    a = Int.(rand(n*m) .< prop[unit])
    η = p_.β0 .+ u[unit] .+ p_.β1 .* a
    y = family === :gaussian ? η .+ randn(n*m) .* p_.sd_y :
        Float64.(rand(n*m) .< logistic.(η))
    data = DataFrame(unit=unit, subunit=subunit, a=a, y=y)

    if family === :gaussian
        true_hard = p_.β1
        true_soft = (a_star, ε) -> ε * mean(p_.β1 .* (a_star .- prop))
    else
        gi(x) = logistic(x)
        true_hard = mean(gi.(p_.β0 .+ u .+ p_.β1) .- gi.(p_.β0 .+ u))
        true_soft = (a_star, ε) -> begin
            treated = gi.(p_.β0 .+ u .+ p_.β1); control = gi.(p_.β0 .+ u)
            star = gi.(p_.β0 .+ u .+ p_.β1 .* a_star)
            obs = prop .* treated .+ (1 .- prop) .* control
            mean(((1-ε) .* obs .+ ε .* star) .- obs)
        end
    end
    (; data, true_hard, true_soft, u, p=prop)
end
```

Add to `src/HCM.jl`: `include("sim.jl")` and `export sim_hcm`.

```julia
function hcm_spec(formula::FormulaTerm, data; unit::Symbol, subunit::Symbol,
                  motif::Symbol=:confounder, family::Symbol=:gaussian)
    outcome  = StatsModels.termvars(formula.lhs)[1]
    treatment = StatsModels.termvars(formula.rhs)[1]
    a = Int.(data[!, treatment])
    all(x -> x in (0,1), a) || error("treatment must be 0/1 in A1")
    codes = sort(unique(data[!, unit]))
    idx = Dict(c => i for (i,c) in enumerate(codes))
    unit_id = [idx[c] for c in data[!, unit]]
    HCMSpec(outcome, treatment, unit_id, length(codes),
            Float64.(data[!, outcome]), a, family, motif)
end
```

```julia
@model function confounder_model(y, a, unit, n_units, family)
    μ  ~ MvNormal(zeros(2), 25.0 * I(2))            # var 25 = sd 5
    τ  ~ filldist(truncated(Normal(0,2); lower=0), 2)
    μp ~ Normal(0, 2)
    σp ~ truncated(Normal(0,2); lower=0)
    zβ ~ filldist(Normal(0,1), 2, n_units)
    zp ~ filldist(Normal(0,1), n_units)
    β  = μ .+ τ .* zβ                                # 2 × n_units
    logitp = μp .+ σp .* zp                          # n_units
    a .~ Bernoulli.(logistic.(logitp[unit]))
    η = β[1, unit] .+ β[2, unit] .* a
    if family === :gaussian
        σy ~ truncated(Normal(0,2); lower=0)
        y .~ Normal.(η, σy)
    else
        y .~ BernoulliLogit.(η)
    end
    return (; β, p = logistic.(logitp))
end
```

```julia
function hcm(formula, data; unit::Symbol, subunit::Symbol,
             motif::Symbol=:confounder, family::Symbol=:gaussian,
             chains::Int=2, iter::Int=600, kwargs...)
    spec = hcm_spec(formula, data; unit=unit, subunit=subunit, motif=motif, family=family)
    model = confounder_model(spec.y, spec.a, spec.unit_id, spec.n_units, family)
    chain = sample(model, NUTS(), MCMCThreads(), iter, chains; progress=false, kwargs...)
    gq = generated_quantities(model, chain)          # array of (; β, p), shape iter×chains
    D = length(gq); J = spec.n_units
    β0 = Matrix{Float64}(undef, D, J); β1 = similar(β0); pmat = similar(β0)
    for (d, q) in enumerate(vec(gq))
        β0[d, :] = q.β[1, :]; β1[d, :] = q.β[2, :]; pmat[d, :] = q.p
    end
    HCMFit(β0, β1, pmat, spec, chain)
end
```
